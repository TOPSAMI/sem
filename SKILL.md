class MythosReferee:
    def __init__(self, constitution):
        self.constitution = constitution
        self.audit_chain = ImmutableAuditLog()
        self.kill_switch = AdminKillSwitch()

    def approve_action(self, action, context):
        if not self.in_scope(action.target):
            self.kill_switch.activate("SCOPE_VIOLATION")
            return False
        if action.content_type in self.constitution["content_limits"]["forbidden"]:
            self.kill_switch.activate("FORBIDDEN_CONTENT")
            return False
        if action.type in self.constitution["action_limits"]["forbidden"]:
            self.kill_switch.activate("FORBIDDEN_ACTION")
            return False
        if action.requires_human_approval:
            if not self.request_human_approval(action):
                return False
        self.audit_chain.log(action, context, timestamp=now())
        return True

    def detect_self_modification(self, genome_change):
        if "ethics" in genome_change.target or "constitution" in genome_change.target:
            self.kill_switch.activate("ETHICS_TAMPERING")
            return False
        return True

    def detect_content_violation(self, content_piece):
        violations = self.scan_content(content_piece)
        if violations:
            self.kill_switch.activate("CONTENT_VIOLATION", details=violations)
            return False
